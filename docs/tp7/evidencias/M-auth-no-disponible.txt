# Se detiene auth-service DESPUÉS de emitir el token.

$ docker compose stop auth-service 2>&1 | tail -n 1
 Container torque-auth-service-1 Stopped 

# El token ya emitido sigue siendo válido: los servicios verifican la firma localmente, sin consultar a auth-service.

$ pedir "$T_MEC1" GET /api/vehiculos/$VEH_COROLLA | tail -n 1
HTTP 200


$ pedir "$T_MEC1" POST /api/ordenes "{\"vehiculoId\":\"$VEH_COROLLA\",\"descripcion\":\"Con auth caído\"}" | tail -n 1
HTTP 201

# Lo que no se puede es obtener tokens nuevos: login → 503 controlado del gateway

$ pedir '' POST /api/auth/login '{"email":"mecanico1@torque.local","password":"clave-de-prueba-1"}'
{"error":"Servicio no disponible.","servicio":"auth-service","causa":"dns","detalles":["auth-service no respondió (dns: EAI_AGAIN). Reintentar más tarde."]}
HTTP 503


$ docker compose start auth-service 2>&1 | tail -n 1
 Container torque-auth-service-1 Started 


$ pedir '' POST /api/auth/login '{"email":"mecanico1@torque.local","password":"clave-de-prueba-1"}' | tail -n 1
HTTP 200
