# Requisito 1: no hay registro abierto. El ADMIN da de alta a los mecánicos.
# Estado inicial: solo el admin (se borran usuarios y sesiones de corridas anteriores).

$ docker compose exec -T auth-db mongosh auth --quiet --eval 'db.sesiones.deleteMany({}); db.usuarios.deleteMany({rol: "MECANICO"}).deletedCount'
0

# El registro público ya no existe → 404

$ pedir '' POST /api/auth/register '{"email":"mecanico1@torque.local","nombre":"Mecanico Uno","password":"clave-de-prueba-1"}'
{"error":"Ruta no encontrada."}
HTTP 404

# Alta sin token → 401

$ pedir '' POST /api/auth/usuarios '{"email":"mecanico1@torque.local","nombre":"Mecanico Uno","password":"clave-de-prueba-1"}'
{"error":"No autenticado.","detalles":["Falta el header Authorization: Bearer <token>."]}
HTTP 401

# Un MECANICO no puede dar de alta usuarios → 403

$ pedir "$T_MEC1" POST /api/auth/usuarios "{\"email\":\"intruso@torque.local\",\"nombre\":\"Intruso\",\"password\":\"$PASS_MEC\",\"rol\":\"ADMIN\"}"
{"error":"Permisos insuficientes.","detalles":["Esta operación requiere el rol ADMIN."]}
HTTP 403
