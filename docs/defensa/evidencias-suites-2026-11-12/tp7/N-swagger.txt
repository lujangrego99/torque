# Swagger agregado en http://localhost:3100/api-docs (selector: vehicles, inventory, workorders, auth, gateway).
# Esquema Bearer y operaciones protegidas con su rol y respuestas 401/403:

$ curl -s http://localhost:3100/api-docs/auth.json | resumen_spec
auth-service · securitySchemes: bearerAuth
  [público] GET    /health                                          Estado del servicio y de su base
  [público] POST   /api/auth/login                          401 403 Login (público) — emite access token (5 min) y refresh token
  [público] POST   /api/auth/refresh                        401     Renovar la sesión (público, con el refresh token) — rota el refresh token
  [público] POST   /api/auth/logout                                 Cerrar sesión (público, con el refresh token) — revoca la renovación
  [bearer]  GET    /api/auth/me                             401     Perfil del usuario del token (MECANICO, ADMIN)
  [bearer]  POST   /api/auth/usuarios                       401 403 Alta de usuario (alta administrada) — rol ADMIN
  [bearer]  GET    /api/auth/usuarios                       401 403 Listar usuarios — rol ADMIN
  [bearer]  PATCH  /api/auth/usuarios/{id}                  401 403 Cambiar rol o habilitar/deshabilitar — rol ADMIN
